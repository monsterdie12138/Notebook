    def compute_distances_two_loops(self, X):
    num_test = X.shape[0]
    num_train = self.X_train.shape[0]
    dists = np.zeros((num_test, num_train))
    for i in range(num_test):
        for j in range(num_train):
            dists[i, j] = np.sqrt(np.sum((X[i] - self.X_train[j]) ** 2))
    return dists

    def compute_distances_one_loop(self, X):
    num_test = X.shape[0]
    num_train = self.X_train.shape[0]
    dists = np.zeros((num_test, num_train))
    for i in range(num_test):
        dists[i, :] = np.sqrt(np.sum((self.X_train - X[i]) ** 2, axis = 1)) # "axis = 1" means we sum along rows
    return dists

    def compute_distances_no_loops(self, X):
    num_test = X.shape[0]
    num_train = self.X_train.shape[0]
    dists = np.zeros((num_test, num_train))
    test_square = np.sum(X ** 2, axis = 1, keepdims = True)             # "keepdims = True" means it is still a 2D array
    train_square = np.sum(self.X_train ** 2, axis = 1, keepdims = True)
    test_dot_train = np.dot(X, self.X_train.T)                          # "T" means transpose
    dists = np.sqrt(test_square + train_square.T - 2 * test_dot_train)  # broadcasting
    return dists

    def predict_labels(self, dists, k=1):
    num_test = dists.shape[0]
    y_pred = np.zeros(num_test)
    for i in range(num_test):
        closest_y = []
        k_indices = np.argsort(dists[i])[:k]                                    # for each test image, find the indices of the k closest training images
        closest_y = self.y_train[k_indices]                                     # get the labels of the k closest training images
        pred_labels, pred_counts = np.unique(closest_y, return_counts = True)   # count the occurrences of each label
        y_pred[i] = pred_labels[np.argmax(pred_counts)]                         # find the label with the highest count as the prediction
    return y_pred

    Q2: Softmax Classifier

    Compute the Gradient of the Loss Function

    $f_j=xW$: the score of class $j$ for a given image $x$

    $p_i=\frac{e^{f_k}}{\sum\limits_{j=1}^Ce^{f_j}}$: the probability of image $i$ to be the right class $k$

    $L_i=-\log p_i=-\log(\frac{e^{f_k}}{\sum\limits_{j=1}^Ce^{f_j}})$: the loss for image $i$

    $L=\frac{1}{N}\sum\limits_{i=1}^NL_i+\lambda W^2$: the total loss for all images, and $\lambda||W||^2$ is the regularization term

    Here, if we want to compute the gradient of the loss with respect to the weights $W$:

    $$dW=\frac{\partial L}{\partial W}$$

    First, we get:

    $$dW=\frac{1}{N}\sum\limits_{i=1}^N\frac{\partial L_i}{\partial W}+2\lambda W$$

    Then we consider $\frac{\partial L_i}{\partial W}$. Using the chain rule, we have:

    $$\frac{\partial L_i}{\partial W}=\sum\limits_{j=1}^C\frac{\partial L_i}{\partial f_j}\frac{\partial f_j}{\partial W}$$

    Consider the first term:

    $$\frac{\partial L_i}{\partial f_j}=\begin{cases} p_j-1,~j=k\ p_j,~j\neq k \end{cases}$$

    Consider the second term:

    $$\frac{\partial f_j}{\partial W_{m,n}}=\begin{cases} x_m,~n=j\ 0,~n\neq j \end{cases}$$

    $$\frac{\partial f_j}{\partial W_{:,n}}=\begin{cases} x,~n=j\ 0,~n\neq j \end{cases}$$

    $$\frac{\partial f_j}{\partial W}=\begin{bmatrix} 0&...&0&x_1&0&...&0\\ 0&...&0&x_2&0&...&0\\ \vdots&\vdots&\vdots&\vdots&\vdots&\vdots&\vdots\\ 0&...&0&x_n&0&...&0 \end{bmatrix}$$

    The column vector $x$ is at the $j$-th column, while all other columns are zero.

    In conclusion, we have:

    $$\frac{\partial L_i}{\partial W}=\begin{bmatrix} p_1x_1&...&p_{k-1}x_1&(p_k-1)x_1&p_{k+1}x_1&...&p_Cx_1\\ p_1x_2&...&p_{k-1}x_2&(p_k-1)x_2&p_{k+1}x_2&...&p_Cx_2\\ \vdots&\vdots&\vdots&\vdots&\vdots&\vdots&\vdots\\ p_1x_n&...&p_{k-1}x_n&(p_k-1)x_n&p_{k+1}x_n&...&p_Cx_n \end{bmatrix}$$

    def softmax_loss_naive(W, X, y, reg):

    # initialize the loss and gradient to zero
    loss = 0.0
    dW = np.zeros_like(W)

    # compute the loss and the gradient
    num_classes = W.shape[1]
    num_train = X.shape[0]

    for i in range(num_train):
        scores = X[i].dot(W)

        # compute the loss in numerically stable way
        scores -= np.max(scores)
        p = np.exp(scores)
        p /= p.sum()
        logp = np.log(p)
        loss -= logp[y[i]]

        # compute the gradient
        for j in range(num_classes):
            if j == y[i]:
                dW[:, j] += (p[j] - 1) * X[i]
            else:
                dW[:, j] += p[j] * X[i]

    loss = loss / num_train + reg * np.sum(W * W)
    dW = dW / num_train + reg * 2 * W
    return loss, dW

    def softmax_loss_vectorized(W, X, y, reg):

    # initialize the loss and gradient to zero
    loss = 0.0
    dW = np.zeros_like(W)

    # compute the loss
    num_classes = W.shape[1]
    num_train = X.shape[0]
    scores = X.dot(W)
    scores -= np.max(scores, axis = 1).reshape(-1, 1)
    p = np.exp(scores)
    p /= np.sum(p, axis = 1).reshape(-1, 1)
    logp = np.log(p)
    loss = -np.sum(logp[np.arange(num_train),y])    # fancy indexing
    loss = loss / num_train + reg * np.sum(W * W)

    # compute the gradient
    p_new = p.copy()
    p_new[np.arange(num_train), y] -= 1
    dW = X.T.dot(p_new)
    dW = dW / num_train + reg * 2 * W

    return loss, dW

    def train(
        self,
        X,
        y,
        learning_rate=1e-3,
        reg=1e-5,
        num_iters=100,
        batch_size=200,
        verbose=False,
    ):
    """
    Inputs:
    - X: A numpy array of shape (N, D) containing training data; there are N
      training samples each of dimension D.
    - y: A numpy array of shape (N,) containing training labels; y[i] = c
      means that X[i] has label 0 <= c < C for C classes.
    - learning_rate: (float) learning rate for optimization.
    - reg: (float) regularization strength.
    - num_iters: (integer) number of steps to take when optimizing
    - batch_size: (integer) number of training examples to use at each step.
    - verbose: (boolean) If true, print progress during optimization.

    Outputs:
    A list containing the value of the loss function at each training iteration.
    """
    num_train, dim = X.shape
    num_classes = (
        np.max(y) + 1
    )                                                                       # assume y takes values 0...K-1 where K is number of classes
    if self.W is None:
        self.W = 0.001 * np.random.randn(dim, num_classes)                  # lazily initialize W

    # Run stochastic gradient descent to optimize W
    loss_history = []
    for it in range(num_iters):
        indices = np.random.choice(num_train, batch_size, replace = True)   # random choice
        X_batch = X[indices]
        y_batch = y[indices]

        loss, grad = self.loss(X_batch, y_batch, reg)                       # compute loss and gradient
        loss_history.append(loss)
        self.W -= grad * learning_rate

        if verbose and it % 100 == 0:
            print("iteration %d / %d: loss %f" % (it, num_iters, loss))

    return loss_history

    Q3: Two-Layer Neural Network

    Two-layer network has a fixed structure.

    • First layer: fully-connected layer + activation function
    • Second layer: fully-connected layer + softmax function

    Affine Layer

    First, let's take a look at fully-connected layer, which is also known as affine layer.

    Consider the forward progress, we have three inputs:

    • $x$: data of shape $(N, d_1, ..., d_k)$, and it should be reshaped to $(N, D)$ where $D = d_1 \cdot ... \cdot d_k$
    • $w$: weights of shape $(D, M)$
    • $b$: biases of shape $(M,)$

    They are used to compute the output $o$ of shape $(N, M)$:

    $$o = x \cdot w + b$$

    def affine_forward(x, w, b):
    N = x.shape[0]
    x_flatten = x.reshape(N, -1)
    out = x_flatten.dot(w) + b
    cache = (x, w, b)   # preserve the inputs
    return out, cache

    When it comes to the backward progress, we need to compute the gradient. If the upstream gradient is dout, we have:

    $$\frac{\partial L}{\partial x} = \frac{\partial L}{\partial o} \cdot \frac{\partial o}{\partial x} = dout \cdot w^T$$

    $$\frac{\partial L}{\partial w} = \frac{\partial L}{\partial o} \cdot \frac{\partial o}{\partial w} = x^T \cdot dout$$

    $$\frac{\partial L}{\partial b} = \frac{\partial L}{\partial o} \cdot \frac{\partial o}{\partial b} = dout$$

    def affine_backward(dout, cache):
    x, w, b = cache
    N = x.shape[0]
    x_flatten = x.reshape(N, -1)
    dx = dout.dot(w.T).reshape(x.shape) # x has been reshaped in the forward pass
    dw = x_flatten.T.dot(dout)
    db = dout.sum(axis = 0)             # b has been broadcast in the forward pass
    return dx, dw, db

    ReLU Activation

    $$\text{ReLU}(x) = \max(0, x)$$

    In the forward pass, ReLU activation function is applied element-wise to the input $x$, which is of any shape.

    def relu_forward(x):
    out = np.maximum(x, 0)
    cache = x
    return out, cache

    In the backward pass, we need to compute the gradient as well.

    def relu_backward(dout, cache):
    dx, x = None, cache
    local_gradient = (x > 0).astype(int) # type casting
    dx = dout * local_gradient
    return dx

    Softmax Layer

    After affine layer 1 + ReLU activation + affine layer 2, we have scores for $C$ categories for $N$ images.

    The last layer is the softmax layer, which is used to compute the loss and the gradient. We take an input $x$ of shape $(N, C)$ and corresponding label $y$. We can find the code is similar to the one we implemented in Q2.

    def softmax_loss(x, y):

    # compute the loss
    N = x.shape[0]
    scores = x.copy()
    scores -= np.max(scores, axis = 1).reshape(-1, 1)
    p = np.exp(scores)
    p /= np.sum(p, axis = 1).reshape(-1, 1)
    logp = np.log(p)
    loss = -np.sum(logp[np.arange(N), y])
    loss = loss / N

    # compute the gradient
    p_new = p.copy()
    p_new[np.arange(N), y] -= 1
    dx = p_new / N

    return loss, dx

    Two-Layer Network

    Since we have implemented the affine layer, ReLU activation, and softmax layer, we can now build a simple two-layer network.

    The class TwoLayerNet has the following data members:

    • self.params: a dictionary containing the weights and biases of the two layers
    • self.reg: a scalar giving the L2 regularization strength

    __init__: The most important part is to initialize the weights and biases. 

    def __init__(
        self,
        input_dim=3 * 32 * 32,  # size of input image
        hidden_dim=100,         # size of intermediate param after first affine layer
        num_classes=10,         # number of classes to classify
        weight_scale=1e-3,      # determine the random initialization of weights
        reg=0.0,                # L2 regularization strength
    ):

        self.params = {}
        self.reg = reg

        self.params['W1'] = np.random.normal(loc=0.0, scale=weight_scale, size=(input_dim, hidden_dim)) # initialize weights from a Gaussian distribution
        self.params['b1'] = 0
        self.params['W2'] = np.random.normal(loc=0.0, scale=weight_scale, size=(hidden_dim, num_classes))
        self.params['b2'] = 0

    loss: The input X is of shape (N, d_1, ..., d_k) and the label y. There are two cases:

    • X is training data: In this case, y is of shape (N,) and this method returns loss and gradients.
    • X is test data: In this case, y is None and this method returns scores of shape (N, C). 
      def loss(self, X, y=None):

        inter_param, inter_cache = affine_relu_forward(X, self.params['W1'], self.params['b1'])
        scores, cache = affine_forward(inter_param, self.params['W2'], self.params['b2'])

        if y is None:
            return scores

        grads = {}
        loss, dscores = softmax_loss(scores, y)
        loss += self.reg / 2 * (np.sum(self.params['W1'] ** 2) + np.sum(self.params['W2'] ** 2))    # L2 regularization is 0.5 * reg * ||W||^2, which is designed to simplify the expression for the gradient
        dinter, grads['W2'], grads['b2'] = affine_backward(dscores, cache)
        _, grads['W1'], grads['b1'] = affine_relu_backward(dinter, inter_cache)
        grads['W1'] += self.reg * self.params['W1'] # do not forget gradient in L2 regularization
        grads['W2'] += self.reg * self.params['W2']

        return loss, grads
